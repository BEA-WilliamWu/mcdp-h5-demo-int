const cp = require("child_process");
const path = require("path");
const CLIEngine = require("eslint").CLIEngine;
const root = cp.execFileSync("git", ["rev-parse", "--show-toplevel"], {encoding:"utf8"}).trim();
// Set BASE to the available pre-HTH baseline in the target repository.
const base = process.env.BASE || "2974edbf";
const files = cp.execFileSync("git", ["diff", "--name-only", "--diff-filter=ACMR", base, "--", "consulting/channel"],
    {cwd:root, encoding:"utf8"}).trim().split("\n").filter(p => p.endsWith(".js")).map(p => path.join(root,p));
if (!files.length) throw new Error("No JavaScript files selected");
const cli = new CLIEngine({configFile:path.resolve(".eslintrc.json"), useEslintrc:false,
    rulePaths:[path.resolve("obdx-eslint-custom-rules")], cache:false, fix:false});
const report = cli.executeOnFiles(files);
console.log(CLIEngine.getFormatter("stylish")(report.results));
console.log({files:report.results.length, errors:report.errorCount, warnings:report.warningCount});
process.exitCode = report.errorCount ? 1 : 0;
